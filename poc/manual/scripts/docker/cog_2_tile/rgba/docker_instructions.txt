# ============================================================
# PUSH EXISTING DOCKER HUB IMAGE TO AWS ECR
# Docker Hub:
#   cfurl/02-pmtile-single-day:v1
#
# ECR:
#   stg4-24hr-large-area-render:02-pmtile-single-day-v1
# ============================================================

$ErrorActionPreference = "Stop"

# ------------------------------------------------------------
# AWS SETTINGS
# ------------------------------------------------------------

$Profile = "stg4-prac"
$Region  = "us-east-2"

$AccountId = aws sts get-caller-identity `
    --query Account `
    --output text `
    --profile $Profile

$EcrHost = "$AccountId.dkr.ecr.$Region.amazonaws.com"

$Repo = "stg4-24hr-large-area-render"

# ------------------------------------------------------------
# IMAGE SETTINGS
# ------------------------------------------------------------

$DockerHubImage = "cfurl/02-pmtile-single-day:v1"

$EcrTag = "02-pmtile-single-day-v1"

$EcrImage = "$EcrHost/${Repo}:$EcrTag"

# ------------------------------------------------------------
# LOGIN TO ECR
# ------------------------------------------------------------

aws ecr get-login-password `
    --region $Region `
    --profile $Profile |
    docker login `
        --username AWS `
        --password-stdin $EcrHost

# ------------------------------------------------------------
# TAG LOCAL DOCKER HUB IMAGE FOR ECR
# ------------------------------------------------------------

docker tag `
    $DockerHubImage `
    $EcrImage

# ------------------------------------------------------------
# PUSH TO ECR
# ------------------------------------------------------------

docker push $EcrImage

# ------------------------------------------------------------
# SHOW RESULT
# ------------------------------------------------------------

Write-Host ""
Write-Host "Pushed to ECR:"
Write-Host $EcrImage
